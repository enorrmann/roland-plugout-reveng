# bq_names_host.txt: names for the lifted Boutique voice with the host's envelopes and LFO (make_voice.py --host,
# README.md §8). Hand-written; bq_names.txt is the version with Roland's own envelopes and LFO.
#
#   in / out ADDR NAME       bank-5 inputs / outputs (out 203A: the voice, bank 1, read by the post program)
#   const ADDR[-ADDR] VALUE  inputs that are constants in bq.c
#   unused ADDR...           outputs bq.c ignores

# per-voice inputs from the ARM (bank 5 A080..A09B, voice A of chip 0)
in A080 pitch
in A081 gate
# the host's envelopes and LFO in the cells of Roland's (make_voice.py --host: dsplift --cell-in)
in A0F0 env_filter
in A0F1 env_amp
in A0F2 lfo_filter
# make_voice.py --osc2x: the oscillators' outputs of the sample before (bq_osc2x.h)
in A0F3 osc1_out
in A0F4 osc2_out
# the pulse width's source: C054 x A090 (bq.c: s1_color(1), OSC2 COLOR x its COLOR MOD source)
in A090 pwm
const A091-A093 0
const A082 0
const A083 0
# envelope scale and polarity: the program uses env^2 x -0.09999484 x (1 - 2 x this); 5.5 gives env^2 x 0.99995
# (positive, full scale; 1 would give x 0.1: envelopes ten times too weak, measured 2026-11-11 on the lift)
const A084 5.5
const A086 5.5
# envelope key follow (ENV2 KEY FOLLOW): 0
const A085 0
const A087 0
# oscillator frequency factors: 2^range per note, 1 = 8' (A4 = 440 Hz), x 44100 / 96000 (README.md §2). (0.5 was a
# compensation for A08B / A08D = 0, which ran the phase over half a ramp: an octave up, half the saw)
const A088 0.459375
const A089 0.459375
# A08B / A08D: the oscillators' phase at a wrap: 1 keeps the remainder (free running), 0 restarts it at the
# sub-sample (sync): 0 made the saw sharp at high notes (C7 +33 cents, measured 2026-11-11)
const A08A 0
const A08B 1
const A08C 0
const A08D 1
const A08E-A08F 0
const A094 1
const A095 0

const A099 0
const A09A 1
const A09B 0
# pre program (LFO): A040 / A041 (LFO gates) 1; A105, A107, A10A (from chip 1) 0
const A040 1
const A041 1
const A105 0
const A107 0
const A10A 0
